\noindent Paesi: panel $i \in \{\text{DE, IT, NL, ES, FR, BE}\}$; serie storiche
per DE, IT, NL, ES, FR, BE, EA. Frequenza trimestrale; orizzonte $h = 0,\dots,12$;
$p = 4$ ritardi.

\subsection{Local Projections: risposte medie}

\noindent Panel (effetti fissi paese, pesi $w_i$, errori standard Driscoll--Kraay con $h+1$ ritardi):
\begin{equation}
y_{i,t+h} = \alpha_i^h + \beta^h s_{i,t}
 + \sum_{l=1}^{p} \left( \gamma_l^h y_{i,t-l} + \delta_l^h s_{i,t-l} \right)
 + \varepsilon_{i,t+h},
\qquad w_i = \frac{1}{T_i} \sum_{t} \mathit{occP}_{i,t}
\end{equation}

\noindent Serie storiche, un paese alla volta (errori standard Newey--West con $h+1$ ritardi):
\begin{equation}
y_{t+h} = \alpha^h + \beta^h s_{t}
 + \sum_{l=1}^{p} \left( \gamma_l^h y_{t-l} + \delta_l^h s_{t-l} \right)
 + \varepsilon_{t+h}
\end{equation}

\noindent Grafici: $\hat\beta^h$ per $h=0,\dots,12$, bande al 68\% e al 90\%.

\subsection{Local Projections: alta e bassa crescita allo shock}

\noindent Stato allo shock:
\begin{equation}
HG_{i,t} = \mathbb{1}\left[ g_{i,t} > \bar g_i \right] \cdot
 \mathbb{1}\left[ g_{i,t+1} > 0 \right] \cdot
 \mathbb{1}\left[ g_{i,t+2} > 0 \right],
\qquad \bar g_i = \text{media storica di } g_{i,t}
\end{equation}

\noindent Specificazione completamente interagita (panel con effetti fissi e pesi $w_i$;
serie storiche analoghe per paese):
\begin{equation}
\begin{aligned}
y_{i,t+h} = \; & \alpha_i^h + \theta^h HG_{i,t} \\
 & + HG_{i,t} \Big[ \beta_H^h s_{i,t} + \sum_{l=1}^{p} \left( \gamma_{H,l}^h y_{i,t-l} + \delta_{H,l}^h s_{i,t-l} \right) \Big] \\
 & + (1-HG_{i,t}) \Big[ \beta_L^h s_{i,t} + \sum_{l=1}^{p} \left( \gamma_{L,l}^h y_{i,t-l} + \delta_{L,l}^h s_{i,t-l} \right) \Big]
 + \varepsilon_{i,t+h}
\end{aligned}
\end{equation}

\noindent Grafici: $\hat\beta_H^h$ (rosso) e $\hat\beta_L^h$ (blu tratteggiato), bande al 90\%.
