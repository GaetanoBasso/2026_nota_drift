\subsection{Local Projections: risposte medie}

\noindent Panel (effetti fissi paese, pesi $w_i$, errori standard Driscoll--Kraay con $h+1$ ritardi):
\begin{equation}
y_{i,t+h} = \alpha_i^h + \beta^h s_{i,t}
 + \sum_{l=1}^{p} \left( \gamma_l^h y_{i,t-l} + \delta_l^h s_{i,t-l} + \phi_l^{h\prime} x_{i,t-l} \right)
 + \kappa^h C_t + \varepsilon_{i,t+h},
\qquad w_i = \frac{1}{T_i} \sum_{t} \mathit{occP}_{i,t}
\end{equation}

\noindent Serie storiche, un paese alla volta (errori standard Newey--West con $h+1$ ritardi):
\begin{equation}
y_{t+h} = \alpha^h + \beta^h s_{t}
 + \sum_{l=1}^{p} \left( \gamma_l^h y_{t-l} + \delta_l^h s_{t-l} + \phi_l^{h\prime} x_{t-l} \right)
 + \kappa^h C_t + \varepsilon_{t+h}
\end{equation}

\noindent Grafici: $\hat\beta^h$ per $h=0,\dots,12$, bande al 68\% e al 90\%.

\subsection{Local Projections: alta e bassa inflazione allo shock}

\noindent Stato allo shock, come in Corsello e Foschi (2026): con $\pi_{i,m}$ la variazione
\% congiunturale mensile dell'HICP core e $m_t$ il primo mese del trimestre $t$,
\begin{equation}
\bar\pi_{i,t} = \frac{1}{6} \sum_{\tau = m_t - 6}^{m_t - 1} \pi_{i,\tau},
\qquad
HI_{i,t} = \mathbb{1}\left[ \bar\pi_{i,t} > P_{75}\big(\bar\pi_{i}\big) \right]
\end{equation}
dove $P_{75}(\bar\pi_{i})$ è il 75° percentile di $\bar\pi_{i,t}$ del paese $i$ sull'intero periodo:
il regime di alta inflazione è un trimestre preceduto da inflazione di fondo elevata.

\noindent Specificazione completamente interagita (panel con effetti fissi e pesi $w_i$;
serie storiche analoghe per paese), con $z_{i,t-l} = (y_{i,t-l}, s_{i,t-l}, x_{i,t-l}')'$:
\begin{equation}
\begin{aligned}
y_{i,t+h} = \; & \alpha_i^h + \theta^h HI_{i,t} \\
 & + HI_{i,t} \Big[ \beta_H^h s_{i,t} + \sum_{l=1}^{p} \psi_{H,l}^{h\prime} z_{i,t-l} + \kappa_H^h C_t \Big] \\
 & + (1-HI_{i,t}) \Big[ \beta_L^h s_{i,t} + \sum_{l=1}^{p} \psi_{L,l}^{h\prime} z_{i,t-l} + \kappa_L^h C_t \Big]
 + \varepsilon_{i,t+h}
\end{aligned}
\end{equation}

\noindent Grafici: $\hat\beta_H^h$ (rosso) e $\hat\beta_L^h$ (blu tratteggiato), bande al 90\%.

\subsection{Robustezza}

\noindent Per l'unità dei grafici principali: specificazione principale con effetti fissi
trimestrali $\sum_{q=2}^{4} \mu_q^h \mathbb{1}[\text{trimestre}(t)=q]$.

\end{itemize}
