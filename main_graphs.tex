\noindent Quarterly frequency (monthly series are averages over complete quarters);
horizon $h = 0,\dots,12$ quarters; $p = 4$ lags.
Panel: $i \in \{\text{DE, IT, NL, ES, FR, BE}\}$ (the euro area is excluded because it
aggregates the countries); time series for DE, IT, NL, ES, FR, BE and EA.
Main text: responses to oil and gas in the same graph, for the euro area and for the euro
area with the individual countries (consumer prices, negotiated wages, hourly wages).
Appendix: hours worked and unemployment rate; graphs by single energy price (including
electricity, estimated on the panel because a national price is not available for the
euro area); robustness.

\subsection{Local projections}

\noindent Panel (country fixed effects, weights $w_i$, Driscoll--Kraay standard errors with $h+1$ lags):
\begin{equation}
y_{i,t+h} = \alpha_i^h + \beta^h s_{i,t}
 + \sum_{l=1}^{p} \left( \gamma_l^h y_{i,t-l} + \delta_l^h s_{i,t-l} + \phi_l^{h\prime} x_{i,t-l} \right)
 + \kappa^h C_t + \varepsilon_{i,t+h},
\qquad w_i = \frac{1}{T_i} \sum_{t} \mathit{occP}_{i,t}
\end{equation}

\noindent Time series, one country at a time (Newey--West standard errors with $h+1$ lags):
\begin{equation}
y_{t+h} = \alpha^h + \beta^h s_{t}
 + \sum_{l=1}^{p} \left( \gamma_l^h y_{t-l} + \delta_l^h s_{t-l} + \phi_l^{h\prime} x_{t-l} \right)
 + \kappa^h C_t + \varepsilon_{t+h}
\end{equation}

\noindent Graphs: $\hat\beta^h$ for $h=0,\dots,12$, 90\% confidence bands.

\subsection{Robustness}

\noindent For the unit of the single-price graphs: pre-COVID sample, data before 2020q1
only ($t+h \le$ 2019q4), without $C_t$.

\subsection{Sources}

\begin{itemize}
\item HICP: Eurostat, \texttt{prc\_hicp\_midx} (\texttt{CP00}, index 2015=100). Core inflation
 (\texttt{TOT\_X\_NRG\_FOOD}) is downloaded and kept in the dataset but not used in the estimates.
\item Negotiated wages: ECB, indicator of negotiated wage rates (INW), total economy,
 annual growth rates; national source series for DE and FR.
\item Wages, hours worked, employment: Eurostat, quarterly national accounts.
\item Industrial production: Eurostat, \texttt{sts\_inpr\_m} (B--D, seasonally adjusted, 2021=100).
 Unemployment rate: Eurostat, \texttt{une\_rt\_m} (seasonally adjusted, total).
\item 1-year Bund yield: Deutsche Bundesbank, term structure of listed Federal securities
 (Svensson method, residual maturity 1 year), series
 \texttt{BBSIS.M.I.ZST.ZI.EUR.S1311.B.A604.R01XX.R.A.A.\_Z.\_Z.A}.
\item Energy prices: file \texttt{Data\_OIL\_ELE\_GAS.xlsx} (monthly data).
\end{itemize}
