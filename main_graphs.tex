\subsection{Local Projections}

\noindent Panel (effetti fissi paese, pesi $w_i$, errori standard Driscoll--Kraay con $h+1$ ritardi):
\begin{equation}
y_{i,t+h} = \alpha_i^h + \beta^h s_{i,t}
 + \sum_{l=1}^{p} \left( \gamma_l^h y_{i,t-l} + \delta_l^h s_{i,t-l} + \phi_l^{h\prime} x_{i,t-l} \right)
 + \kappa^h C_t + \varepsilon_{i,t+h},
\qquad w_i = \frac{1}{T_i} \sum_{t} \mathit{occP}_{i,t}
\end{equation}

\noindent Serie storiche, un paese alla volta (errori standard Newey--West con $h+1$ ritardi):
\begin{equation}
y_{t+h} = \alpha^h + \beta^h s_{t}
 + \sum_{l=1}^{p} \left( \gamma_l^h y_{t-l} + \delta_l^h s_{t-l} + \phi_l^{h\prime} x_{t-l} \right)
 + \kappa^h C_t + \varepsilon_{t+h}
\end{equation}

\noindent Grafici: $\hat\beta^h$ per $h=0,\dots,12$, bande al 90\%.

\subsection{Robustezza}

\noindent Per l'unità dei grafici per singolo prezzo: campione pre-Covid, solo dati
precedenti il 2020q1 ($t+h \le$ 2019q4), senza $C_t$.

\subsection{Fonti}

\begin{itemize}
\item HICP: Eurostat, \texttt{prc\_hicp\_midx} (\texttt{CP00}, indice 2015=100). L'inflazione
 di fondo (\texttt{TOT\_X\_NRG\_FOOD}) è scaricata e inclusa nel dataset ma non usata nelle stime.
\item Retribuzioni contrattuali: BCE, indicatore delle retribuzioni contrattuali (INW),
 totale economia, tassi di crescita annuali; per DE e FR serie delle fonti nazionali.
\item Retribuzioni, ore lavorate, occupati: Eurostat, conti nazionali trimestrali.
\item Produzione industriale: Eurostat, \texttt{sts\_inpr\_m} (B--D, dest., 2021=100).
 Tasso di disoccupazione: Eurostat, \texttt{une\_rt\_m} (dest., totale).
\item Rendimento del Bund a 1 anno: Deutsche Bundesbank, struttura per scadenza dei
 titoli federali (metodo Svensson, vita residua 1 anno), serie
 \texttt{BBSIS.M.I.ZST.ZI.EUR.S1311.B.A604.R01XX.R.A.A.\_Z.\_Z.A}.
\item Prezzi energetici: file \texttt{Data\_OIL\_ELE\_GAS.xlsx} (dati mensili).
\end{itemize}
